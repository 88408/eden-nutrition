export const useAuthStore = defineStore('auth', {
  state: () => ({
    token: localStorage.getItem('token') || '',
    user: null as User | null
  }),
  getters: {
    isLoggedIn: (state) => !!state.token
  },
  actions: {
    async login(data: LoginDTO) {
      const response = await authApi.login(data)
      this.token = response.token
      localStorage.setItem('token', this.token)
      await this.fetchCurrentUser()
      return response
    },
    async fetchCurrentUser() {
      this.user = await authApi.getCurrentUser()
    },
    logout() {
      this.token = ''
      this.user = null
      localStorage.removeItem('token')
    }
  }
})

export const useCartStore = defineStore('cart', {
  state: () => ({
    items: [] as CartItem[],
    totalCount: 0,
    totalAmount: 0,
    loading: false
  }),
  actions: {
    async fetchCart() {
      this.loading = true
      try {
        this.items = await cartApi.getCartList()
      } finally {
        this.loading = false
      }
    }
  }
})

export const useOrderStore = defineStore('order', {
  state: () => ({
    orders: [] as Order[],
    total: 0,
    loading: false,
    filters: {
      status: undefined as number | undefined,
      pageNum: 1,
      pageSize: 10
    }
  }),
  actions: {
    async fetchOrders() {
      this.loading = true
      try {
        const response = await orderApi.getOrderList(this.filters)
        this.orders = response.list
        this.total = response.total
      } finally {
        this.loading = false
      }
    }
  }
})

router/guards.ts:

export function setupRouterGuards(router: Router) {
  router.beforeEach((to, from, next) => {
    const authStore = useAuthStore()
    document.title = `${(to.meta.title as string) || 'Eden Nutrition'} - Eden Nutrition`
    if (to.meta.requiresAuth && !authStore.isLoggedIn) {
      next('/auth/login')
      return
    }
    next()
  })
}

权限与状态设计
================================================================================

用户端一般不使用复杂 RBAC，只需要:
- 是否登录
- 是否拥有管理员身份（如需要隐藏入口）
- 订单状态、优惠券状态、秒杀状态的展示映射

建议状态映射:
- 订单状态: 0待支付 / 1已支付 / 2已发货 / 3已完成 / 4已取消 / 5已退款
- 优惠券状态: 0未使用 / 1已使用 / 2已过期
- 秒杀状态: 0未开始 / 1进行中 / 2已结束

import axios, { AxiosInstance } from 'axios'

const client: AxiosInstance = axios.create({
  baseURL: import.meta.env.VITE_API_BASE_URL,
  timeout: 10000,
  headers: {
    'Content-Type': 'application/json'
  }
})

client.interceptors.request.use((config) => {
  const authStore = useAuthStore()
  if (authStore.token) {
    config.headers.Authorization = `Bearer ${authStore.token}`
  }
  return config
})

client.interceptors.response.use(
  (response) => {
    const { code, data, message } = response.data
    if (code === 200) {
      return data
    }
    ElMessage.error(message || '操作失败')
    throw new Error(message)
  },
  (error) => {
    if (error.response?.status === 401) {
      const authStore = useAuthStore()
      authStore.logout()
      router.push('/auth/login')
    }
    return Promise.reject(error)
  }
)

export default client
